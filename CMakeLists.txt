cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED ENV{ASCEND_HOME_PATH})
  message(FATAL_ERROR "请先 source CANN 的 set_env.sh")
endif()
set(CANN "$ENV{ASCEND_HOME_PATH}")
set(CMAKE_CXX_COMPILER "${CANN}/bin/bisheng" CACHE FILEPATH "毕昇编译器")
project(ascend_kernel_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(AKL_NPU_ARCH "dav-2201" CACHE STRING "目标架构，本轮验证 A3 dav-2201")
add_library(akl_kernels SHARED kernels/micro.cpp)
target_include_directories(akl_kernels PRIVATE include "${CANN}/include")
target_compile_options(akl_kernels PRIVATE -O2 -fPIC -xasc "--npu-arch=${AKL_NPU_ARCH}")
target_link_options(akl_kernels PRIVATE --cce-fatobj-link)
target_link_directories(akl_kernels PRIVATE "${CANN}/lib64")
target_link_libraries(akl_kernels PRIVATE ascendcl runtime)
set_target_properties(akl_kernels PROPERTIES BUILD_RPATH "${CANN}/lib64")
