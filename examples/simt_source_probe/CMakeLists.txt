cmake_minimum_required(VERSION 3.20)
set(CANN "$ENV{ASCEND_HOME_PATH}")
if(NOT EXISTS "${CANN}/bin/bisheng")
  message(FATAL_ERROR "请先 source 目标 CANN 的 set_env.sh，确认 ASCEND_HOME_PATH")
endif()
set(CMAKE_CXX_COMPILER "${CANN}/bin/bisheng")
project(akl_simt_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(akl_simt_probe main.cpp kernel.cpp)
target_include_directories(akl_simt_probe PRIVATE "${CANN}/include")
target_compile_options(akl_simt_probe PRIVATE -O2 -g)
# 仅设备翻译单元使用 Ascend C；保留优化和行信息，不用 -O0 代替真实映射。
set_source_files_properties(kernel.cpp PROPERTIES COMPILE_OPTIONS "-xasc;--npu-arch=dav-3510")
target_link_options(akl_simt_probe PRIVATE --cce-fatobj-link)
target_link_directories(akl_simt_probe PRIVATE "${CANN}/lib64")
target_link_libraries(akl_simt_probe PRIVATE ascendcl runtime)
set_target_properties(akl_simt_probe PROPERTIES BUILD_RPATH "${CANN}/lib64")
